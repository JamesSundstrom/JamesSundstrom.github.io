  1. Write a function sortByLength that takes as argument a std::vector<std::string> and sorts its elements by length, least to greatest. Strings of the same length should be sorted alphabetically. (You may assume all strings consist only of lowercase letters, so that "alphabetically" matches the default ordering of strings.) Here's some code to help you test your function: 
    #include <iostream>
#include <vector>
#include <string>
#include <algorithm>

void sortByLength(std::vector<std::string> &v) {
    // Your code here
}

int main() {
    std::vector<std::string> v = {"garlic", "fig", "banana", "apple", "kiwi", "eggplant", "honeydew", "date", "cucumber"};
    sortByLength(v);
    for (const std::string &s : v) {
        std::cout << s << ' '; // fig date kiwi apple banana garlic cucumber eggplant honeydew 
    }
    std::cout << std::endl;
    return 0;
}

    Answer: 

    void sortByLength(std::vector<std::string> &v) {
    std::sort(v.begin(), v.end(), [](std::string a, std::string b) 
        {return a.size() < b.size() || (a.size() == b.size() && a < b);});
}

  2. Write a function countAboveAverage that takes as argument a std::vector<int> and returns the number of elements in the vector that are larger than the average of the vector's elements. Can you do it with just two statements? 
    #include <iostream>
#include <vector>
#include <algorithm>

// Your code here

int main() {
    std::vector<int> v1 = {1, 2, 3, 6, 15, 20, 50, 75, 200};
    std::vector<int> v2 = {5, 5, 5, 5};
    std::vector<int> v3 = {5, -1, 3, -7, 15};
    std::cout << countAboveAverage(v1) << std::endl; // 3
    std::cout << countAboveAverage(v2) << std::endl; // 0
    std::cout << countAboveAverage(v3) << std::endl; // 2
    return 0;    
}

    Answer:

    #include <vector>
#include <algorithm>
#include <numeric>

int countAboveAverage(const std::vector<int>& v) {
    int average = std::accumulate(v.begin(), v.end(), 0) / v.size();
    return std::count_if(v.begin(), v.end(), [average](int a) {return a > average;});
}

  3. Using only std::accumulate, write a function toBinary that takes as argument a std::vector<bool> and converts it to a binary number, treating true as 1, and reading left-to-right. For example, toBinary({true, false, false, true, false}) would return 18 (100102). 
    #include <iostream>
#include <vector>
#include <numeric>

// Your code here

int main() {
    std::vector<bool> v = {true, false, false, true, false};
    std::cout << toBinary(v) << std::endl; // 18
    return 0;
}

    4. Answer: 

    #include <vector>
#include <numeric>

int toBinary(const std::vector<bool>& v) {
    return std::accumulate(v.begin(), v.end(), 0, [](int acc, int val) {return 2 * acc + val;});
}
